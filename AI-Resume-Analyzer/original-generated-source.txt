        function analyzeTextLocally(text) {
            const lowerText = text.toLowerCase();
            
            // 1. Detect Technical and Soft Skills
            const detectedSkills = [];
            SKILLS_DICTIONARY.forEach(skill => {
                if (lowerText.includes(skill.toLowerCase())) {
                    detectedSkills.push(skill);
                }
            });

            // Fallback skills if text is very short
            if (detectedSkills.length === 0) {
                detectedSkills.push('JavaScript', 'HTML', 'CSS', 'Communication', 'Problem Solving');
            }

            // Categorize Skills
            const techSkills = detectedSkills.filter(s => !['Communication', 'Problem Solving', 'Leadership', 'Teamwork', 'Agile', 'Product Design', 'UI/UX'].includes(s));
            const softSkills = detectedSkills.filter(s => ['Communication', 'Problem Solving', 'Leadership', 'Teamwork', 'Agile', 'Product Design', 'UI/UX'].includes(s));

            // 2. Section Checks
            const hasEdu = /education|bachelor|master|degree|university|college/i.test(text);
            const hasExp = /experience|work|employment|position|developer|engineer|manager/i.test(text);
            const hasProj = /project|built|developed|created|github/i.test(text);
            const hasContact = /email|phone|linkedin|github|@|\d{10}/i.test(text);

            // 3. Action Verbs Check
            const actionVerbs = ['developed', 'engineered', 'built', 'managed', 'created', 'optimized', 'designed', 'increased', 'reduced', 'spearheaded', 'achieved'];
            let actionVerbCount = 0;
            actionVerbs.forEach(v => {
                if (lowerText.includes(v)) actionVerbCount++;
            });

            // 4. Score Calculations
            const atsScore = Math.min(98, Math.max(55, (hasEdu ? 25 : 10) + (hasExp ? 30 : 10) + (hasContact ? 25 : 5) + (hasProj ? 15 : 5)));
            const skillsScore = Math.min(95, Math.max(50, detectedSkills.length * 7));
            const expScore = hasExp ? 85 : 60;
            const eduScore = hasEdu ? 90 : 65;
            const verbScore = Math.min(95, 50 + (actionVerbCount * 6));
            const overallScore = Math.round((atsScore * 0.3) + (skillsScore * 0.3) + (expScore * 0.2) + (eduScore * 0.1) + (verbScore * 0.1));

            // 5. Candidate Profile Extraction
            let candidateName = 'Candidate';
            const lines = text.split('\n').filter(l => l.trim().length > 0);
            if (lines.length > 0 && lines[0].length < 35 && !lines[0].includes('@')) {
                candidateName = lines[0].trim();
            }

            let candidateRole = 'Software Engineer';
            if (lowerText.includes('data scientist') || lowerText.includes('machine learning')) candidateRole = 'Data Scientist';
            else if (lowerText.includes('ui/ux') || lowerText.includes('designer')) candidateRole = 'UI/UX Product Designer';
            else if (lowerText.includes('full stack')) candidateRole = 'Full Stack Developer';
            else if (lowerText.includes('frontend')) candidateRole = 'Frontend Developer';
            else if (lowerText.includes('backend')) candidateRole = 'Backend Developer';

            // 6. Generate Improvement Items
            const improvementsHigh = [];
            const improvementsMedium = [];
            const improvementsLow = [];

            if (!hasContact) improvementsHigh.push("Add clear contact information (email, phone, LinkedIn profile link).");
            if (actionVerbCount < 3) improvementsHigh.push("Incorporate strong action verbs like 'Engineered', 'Optimized', and 'Spearheaded'.");
            if (detectedSkills.length < 8) improvementsHigh.push("List more specific technical frameworks and modern tech stack keywords.");

            if (!hasProj) improvementsMedium.push("Add a dedicated 'Projects' section featuring live links or GitHub repositories.");
            improvementsMedium.push("Quantify achievements with metrics (e.g., 'improved performance by 30%').");
            improvementsMedium.push("Ensure consistent date formatting across all work history entries.");

            improvementsLow.push("Keep resume layout clean with standard single-column ATS formatting.");
            improvementsLow.push("Verify there are no typographical or spelling errors in technical tool names.");

            return {
                overallScore,
                atsScore,
                skillsScore,
                expScore,
                eduScore,
                verbScore,
                candidateName,
                candidateRole,
                candidateExp: '2 - 4 Years',
                techSkills,
                softSkills: softSkills.length > 0 ? softSkills : ['Communication', 'Problem Solving', 'Teamwork'],
                improvementsHigh,
                improvementsMedium,
                improvementsLow
            };
        }

        async function startResumeAnalysis() {
            document.getElementById('analysis-loading').classList.remove('hidden');
            document.getElementById('analysis-results').classList.add('hidden');

            const progressBar = document.getElementById('analysis-progress-bar');
            const stepText = document.getElementById('loading-step-text');

            // 1. Reading content
            progressBar.style.width = '25%';
            stepText.innerText = "Reading resume content...";

            if (state.selectedFile && !state.extractedText) {
                if (state.selectedFile.type === "application/pdf") {
                    try {
                        const arrayBuffer = await state.selectedFile.arrayBuffer();
                        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
                        let fullText = "";
                        for (let i = 1; i <= pdf.numPages; i++) {
                            const page = await pdf.getPage(i);
                            const textContent = await page.getTextContent();
                            fullText += textContent.items.map(item => item.str).join(" ") + "\n";
                        }
                        state.extractedText = fullText;
                    } catch (err) {
                        console.error("PDF parse error:", err);
                        state.extractedText = "Fallback extracted text from resume.";
                    }
                } else {
                    state.extractedText = await state.selectedFile.text();
                }
            }

            await new Promise(r => setTimeout(r, 600));

            // 2. Extracting skills
            progressBar.style.width = '50%';
            stepText.innerText = "Extracting technical skills & qualifications...";
            await new Promise(r => setTimeout(r, 600));

            // 3. ATS Benchmark calculation
            progressBar.style.width = '75%';
            stepText.innerText = "Calculating ATS score & resume impact...";
            await new Promise(r => setTimeout(r, 600));

            // 4. Finalizing
            progressBar.style.width = '100%';
            stepText.innerText = "Finalizing recommendations...";
            await new Promise(r => setTimeout(r, 400));

            // Run Analysis
            state.analysisData = analyzeTextLocally(state.extractedText || PRESET_RESUMES.fullstack);
            renderAnalysisResults(state.analysisData);

            document.getElementById('analysis-loading').classList.add('hidden');
            document.getElementById('analysis-results').classList.remove('hidden');

            try {
                confetti({ particleCount: 60, spread: 70, origin: { y: 0.6 } });
            } catch(e){}

            showToast('Resume analysis completed!', 'success');
        }

        /* ================= JOB MATCHING ENGINE ================= */
        function calculateJobMatch(job, userSkills) {
            if (!userSkills || userSkills.length === 0) {
                userSkills = ['React', 'JavaScript', 'HTML', 'CSS', 'Node.js', 'Git'];
            }

            const lowerUserSkills = userSkills.map(s => s.toLowerCase());
            const matchingSkills = [];
            const missingSkills = [];

            job.skills.forEach(reqSkill => {
                if (lowerUserSkills.some(us => us.includes(reqSkill.toLowerCase()) || reqSkill.toLowerCase().includes(us))) {
                    matchingSkills.push(reqSkill);
                } else {
                    missingSkills.push(reqSkill);
                }
            });

            const skillMatchPct = matchingSkills.length / job.skills.length;
            const matchScore = Math.min(98, Math.max(45, Math.round(skillMatchPct * 100)));

            return {
                matchScore,
                matchingSkills,
                missingSkills
            };
        }

        function filterAndRenderJobs() {
            const searchQuery = document.getElementById('filter-search').value.toLowerCase();
            const categoryFilter = document.getElementById('filter-category').value;
            const minMatch = parseInt(document.getElementById('filter-min-match').value) || 0;
            const sortBy = document.getElementById('filter-sort').value;

            const userSkills = state.analysisData ? [...state.analysisData.techSkills, ...state.analysisData.softSkills] : ['React', 'JavaScript', 'HTML', 'CSS', 'Node.js', 'Git'];

            let filtered = JOBS_DATABASE.map(job => {
                const matchRes = calculateJobMatch(job, userSkills);
                return { ...job, ...matchRes };
            });

            // Filters
            filtered = filtered.filter(job => {
                const matchesSearch = job.title.toLowerCase().includes(searchQuery) ||
                                      job.company.toLowerCase().includes(searchQuery) ||
                                      job.skills.some(s => s.toLowerCase().includes(searchQuery));
                const matchesCat = categoryFilter === 'ALL' || job.category === categoryFilter;
                const matchesMin = job.matchScore >= minMatch;

                return matchesSearch && matchesCat && matchesMin;
            });

            // Sorting
            if (sortBy === 'match-desc') {
                filtered.sort((a, b) => b.matchScore - a.matchScore);
            } else if (sortBy === 'salary-desc') {
                filtered.sort((a, b) => b.salary.localeCompare(a.salary));
            } else if (sortBy === 'title-asc') {
                filtered.sort((a, b) => a.title.localeCompare(b.title));
            }

            const container = document.getElementById('jobs-grid');
            if (filtered.length === 0) {
                container.innerHTML = `

        /* ================= UTILITY TOAST NOTIFICATIONS ================= */
        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            
            const bgClass = type === 'success' ? 'bg-emerald-500/90' : type === 'error' ? 'bg-red-500/90' : 'bg-purplePrimary/90';
            
            toast.className = `${bgClass} text-white text-xs font-semibold px-4 py-3 rounded-xl shadow-xl backdrop-blur-md flex items-center space-x-2 transition-all transform translate-y-2 opacity-0 pointer-events-auto`;
            toast.innerHTML = `<i class="fa-solid fa-circle-info"></i><span>${message}</span>`;
            
            container.appendChild(toast);
            
            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3500);
        }
